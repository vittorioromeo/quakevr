cmake_minimum_required(VERSION 3.16)
project(qvr_raster_depth_probe LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 gl)
find_package(Vulkan REQUIRED)
find_path(XR_INCLUDE openxr/openxr.h REQUIRED)
find_library(XR_LOADER openxr_loader HINTS /opt/steamvr/bin/linuxarm64 REQUIRED)
find_program(GLSLC glslc)
set(SHADER "${CMAKE_CURRENT_BINARY_DIR}/bridge_depth_readback.comp.spv")
if(GLSLC)
  set(SHADER_COMMAND "${GLSLC}" --target-env=vulkan1.1)
else()
  find_program(GLSLANG glslangValidator REQUIRED)
  set(SHADER_COMMAND "${GLSLANG}" -V --target-env vulkan1.1 -S comp)
endif()
add_custom_command(OUTPUT "${SHADER}"
  COMMAND ${SHADER_COMMAND} "${CMAKE_CURRENT_LIST_DIR}/../shaders/bridge_depth_readback.comp" -o "${SHADER}"
  DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../shaders/bridge_depth_readback.comp" VERBATIM)
add_executable(qvr_raster_depth_probe ../vr_gl_vulkan_depth_raster_test.cpp
  ../../../../Quake/vr/vr_gl_vulkan_bridge.cpp "${SHADER}")
target_compile_features(qvr_raster_depth_probe PRIVATE cxx_std_17)
target_include_directories(qvr_raster_depth_probe PRIVATE "${XR_INCLUDE}" ../../../../Quake/vr)
target_compile_definitions(qvr_raster_depth_probe PRIVATE QVR_HAVE_GL_VULKAN_BRIDGE "QVR_DEPTH_READBACK_SHADER=\"${SHADER}\"")
target_compile_options(qvr_raster_depth_probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_link_libraries(qvr_raster_depth_probe PRIVATE PkgConfig::DEPS Vulkan::Vulkan "${XR_LOADER}")
