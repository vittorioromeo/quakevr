cmake_minimum_required(VERSION 3.16)
project(qvr_vulkan_bootstrap_guards LANGUAGES CXX)
option(QVR_ENABLE_FRAME_SYNTHESIS_GL_VULKAN_BRIDGE "Compile optional Linux Vulkan2 bootstrap" OFF)
add_executable(qvr_vulkan_bootstrap_test ../vr_openxr_vulkan_bootstrap_test.cpp
  ../../../../Quake/vr/vr_openxr_vulkan.cpp)
target_include_directories(qvr_vulkan_bootstrap_test PRIVATE ../../../../Quake/vr)
target_compile_features(qvr_vulkan_bootstrap_test PRIVATE cxx_std_17)
target_compile_options(qvr_vulkan_bootstrap_test PRIVATE -fno-exceptions -Wall -Wextra -Werror -Wno-missing-field-initializers)
if(QVR_ENABLE_FRAME_SYNTHESIS_GL_VULKAN_BRIDGE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Optional Vulkan2 bootstrap requires Linux")
  endif()
  find_package(Vulkan REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GL_DEPS REQUIRED IMPORTED_TARGET sdl2 gl)
  find_path(XR_INCLUDE openxr/openxr.h REQUIRED)
  find_library(XR_LOADER openxr_loader HINTS /opt/steamvr/bin/linuxarm64 REQUIRED)
  target_include_directories(qvr_vulkan_bootstrap_test PRIVATE "${XR_INCLUDE}")
  target_compile_definitions(qvr_vulkan_bootstrap_test PRIVATE QVR_HAVE_GL_VULKAN_BRIDGE)
  target_link_libraries(qvr_vulkan_bootstrap_test PRIVATE Vulkan::Vulkan "${XR_LOADER}" PkgConfig::GL_DEPS)
endif()
enable_testing()
add_test(NAME vulkan_bootstrap_cpu_guards COMMAND qvr_vulkan_bootstrap_test)
