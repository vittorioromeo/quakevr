cmake_minimum_required(VERSION 3.16)
project(qvr_gl_formats_probe LANGUAGES CXX)
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "This opt-in diagnostic uses Linux OPAQUE_FD and GL/Zink")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(PROBE_DEPS REQUIRED IMPORTED_TARGET sdl2 gl vulkan)
find_path(PROBE_XR_INCLUDE openxr/openxr.h REQUIRED)
find_library(PROBE_XR_LOADER openxr_loader HINTS /opt/steamvr/bin/linuxarm64 REQUIRED)
add_executable(qvr_openxr_gl_formats_probe ../vr_openxr_gl_formats_probe.cpp)
target_compile_features(qvr_openxr_gl_formats_probe PRIVATE cxx_std_17)
target_include_directories(qvr_openxr_gl_formats_probe PRIVATE ${PROBE_XR_INCLUDE})
target_link_libraries(qvr_openxr_gl_formats_probe PRIVATE PkgConfig::PROBE_DEPS ${PROBE_XR_LOADER})
target_compile_options(qvr_openxr_gl_formats_probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
