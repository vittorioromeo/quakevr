// quakevr/data/arena_director.txt -- the Survival Arena budget director's costs, curves, themes and supplies

// `<kind> <name> { key "value" ... }`, // comments.
//
// Costs are THREAT points: an unarmed grunt is 4, a shotgun grunt 8, an ogre 25. Waves are Normal's (the difficulty
// moves every `from` past wave 3 as the classic waves' gates: +3 Easy, -2 Hard, -4 Nightmare, by vr_arena_skill_impact).
//   cost   threat points          from   first wave it may appear   until  last wave (0: no end)
//   share  max share of a wave's threat budget it may take            weight base pick weight (before biases)
//   tags   what the themes bias on (humanoid, monster, knight, melee, ranged, far, flying, fast, cheap ...)
//   arms   a body's weapons: any, melee (blades only), guns (anything but blades), none (its own attack)
//   family a weapon's family (One Family): shotguns, bullets, explosives, energy, blades
//   title  its words in a boss's countdown

// The budget: threat(w) = base + slope * (w-1)^power, times the sawtooth's step (the boss cycle's position:
// vr_arena_boss_every waves, the boss wave last), times the skill's factor and vr_arena_wave_size.
// count: a wave's foes at most, a + b * w (times vr_arena_wave_size and the skill's factor); the budget a full wave
// could not place goes to upgrades, then carries over. overshoot: the last foe may overspend by this many points (the
// debt comes off the next wave). boss_share: a boss takes at most this share of its wave (the cheapest when none fits).
// reserve: the share held back for the wave's surges (from reserve_from).
curve threat { base "9" slope "7.5" power "1.05" count "3 0.5" saw "0.85 0.95 1.05 1.15 1.6" carry_max "0.25" reserve "0.2" reserve_from "6" overshoot "2" boss_share "0.55" }
// The roles: a foe's role is its total cost's band. The split: the share of the budget each role gets -- fodder,
// mid, heavy, and extras (the upgrades: armour, a second weapon, infection) -- at key waves (linear in between, the
// last held). humanoid: each pick is from the armed humanoids alone with this chance (when one fits).
curve roles { fodder_max "12" mid_max "24" humanoid "0.7" split "1:1,0,0,0 4:0.78,0.2,0,0.02 8:0.54,0.3,0.1,0.06 14:0.36,0.3,0.17,0.17 22:0.24,0.29,0.2,0.27 30:0.2,0.27,0.2,0.33" }
// Anti-repetition: an item last seen d waves ago weighs recent[d] (d >= 4: 1, +bonus a wave unseen, at most cap);
// an item already in this wave weighs `again` per copy. Old items fade past fade_after waves in play.
curve novelty { recent "0.4 0.6 0.85" bonus "0.1" cap "1.6" again "0.75" fade_after "10" fade "0.05" floor "0.35" }
// Progression: every `every` waves at least one foe carries a weapon no earlier wave offered (lowest tier first;
// up to `peek` waves before its own wave when every unlocked one was offered).
curve progress { every "3" from "3" peek "3" }
// The player's supplies after a clear: points = base + per_threat * threat just beaten, divided by the skill's factor.
// The power-ups' share is banked from clear to clear until it buys one (a gap of `gap` waves between two).
curve supply { base "8" per_threat "0.06" max "60" heal "0.55" armour "0.30" ammo "0.08" power "0.07" reactive "1" }
// The cadence (a theme's own values win): gap s between two spawn groups at Normal (the skill divides it), stretched
// by 1 + stretch x the threat up / the wave's threat, and doubled again past `soft` of it; group: a spawn group's
// foes, min max; surges "a b max": a + b x wave surges a wave (from reserve_from), each warned surge_warn s ahead and
// spread round the player; spread 1: every group from places spread round him; ambush 1: surges behind him first.
// quick_time, quick_pay: a foe killed within this many s of coming in pays this much more (vr_arena_quick_kill).
curve cadence { gap "3" group "1 2" stretch "1.5" soft "0.6" spread "0" surges "1 0.05 3" surge_warn "2" quick_time "4" quick_pay "0.25" }
